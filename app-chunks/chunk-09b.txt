").value;
  const t=team(teamName);
  if(!p){toast("Seleziona un calciatore valido.");return}
  if(soldEntry(p.name)){toast("Calciatore già assegnato.");return}
  if(!t||amount<1){toast("Rilancio non valido.");return}
  if(slotsLeft(t,p.role)<=0){toast(`${t.name}: nessuno slot ${p.role} disponibile.`);return}
  if(amount>teamLegalMax(t,p)){toast(`${t.name} non può arrivare a ${amount} crediti.`);return}
  const prev=lastBidFor(p.name);
  if(prev && amount<=prev.amount){toast(`Il rilancio deve superare ${prev.amount}.`);return}
  if(prev && prev.team===teamName){toast(`${teamName} è già la squadra in testa.`);return}
  state.bidHistory.push({player:p.name,role:p.role,team:teamName,amount,ts:Date.now()});
  renderAll();
  toast(prev?`Controrilancio: ${teamName} a ${amount}`:`Prima offerta: ${teamName} a ${amount}`);
}

function assign(teamName){
  const p=playerByName(byId("auctionPlayer").value);
  const price=+byId("currentBid").value||0;
  if(!p){toast("Seleziona un calciatore valido.");return}
  if(soldEntry(p.name)){toast("Calciatore già assegnato.");return}
  if(price<1){toast("Inserisci un prezzo valido.");return}
  const t=team(teamName);
  if(!t){return}
  if(slotsLeft(t,p.role)<=0){toast(`${t.name}: nessuno slot ${p.role} disponibile.`);return}
  if(price>maxAffordable(t)){toast(`${t.name} deve conservare almeno 1 credito per ogni slot residuo.`);return}
  const last=lastBidFor(p.name);
  if(!last || last.team!==t.name || last.amount!==price){state.bidHistory.push({player:p.name,role:p.role,team:t.name,amount:price,ts:Date.now(),final:true});}
  t.spent+=price;
  t.roster.push({player:p.name,role:p.role,price});
  state.history.push({player:p.name,role:p.role,team:t.name,price,ts:Date.now()});
  byId("auctionPlayer").value="";
  byId("currentBid").value=1;
  renderAll();
}

function undo(){
  const h=state.history.pop();
  if(!h){toast("Nessun acquisto da annullare.");return}
  const t=team(h.team);
  if(t){
    t.spent-=h.price;
    const i=t.roster.findIndex(x=>x.player===h.player);
    if(i>=0)t.roster.splice(i,1);
  }
  renderAll();toast(`Annullato: ${h.player}`);
}

function toggleFav(name){
  if(!name)return;
  const i=state.favorites.indexOf(name);
  if(i>=0)state.favorites.splice(i,1); else state.favorites.push(name);
  renderPlayers();renderStrategy();renderAuction();saveState();
}

function importBackup(file){
  const r=new FileReader();
  r.onload=()=>{
    try{const s=JSON.parse(r.result);if(!s.favorites)s.favorites=[];if(!s.bidHistory)s.bidHistory=[];state=s;renderAll();toast("Backup importato.");}
    catch(e){toast("Backup JSON non valido.");}
  };
  r.readAsText(file);
}
