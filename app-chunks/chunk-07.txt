function inflation(role){
  const vals=[];
  for(const h of state.history){
    if(role && h.role!==role) continue;
    const p=PLAYERS.find(x=>x.name===h.player);
    if(p && p.base600>0) vals.push(h.price/p.base600);
  }
  if(!vals.length) return 0;
  // media robusta: limita l'effetto di singoli acquisti estremi
  const clamped=vals.map(x=>Math.max(.55,Math.min(1.65,x)));
  return clamped.reduce((a,b)=>a+b,0)/clamped.length-1;
}

function dynamicMax(p){
  if(!p) return 0;
  const me=team(state.config.mine);
  if(!me || slotsLeft(me,p.role)<=0) return 0;

  const roleInfl=Math.max(-.15,Math.min(.35,inflation(p.role)));
  const overallInfl=Math.max(-.10,Math.min(.20,inflation()));
  const infl=(roleInfl*0.75)+(overallInfl*0.25);

  const planned=Math.round(state.config.initialCredits*(state.config.budgetPct[p.role]||0)/100);
  const spentRole=me.roster.filter(x=>x.role===p.role).reduce((s,x)=>s+x.price,0);
  const roleLeft=Math.max(0,planned-spentRole);
  const roleReserve=Math.max(0,slotsLeft(me,p.role)-1);
  const roleCap=Math.max(0,roleLeft-roleReserve);

  let dyn=Math.round(p.baseMax*(1+infl));
  // Se il piano di reparto è già stato sforato, non azzera il valore: usa il tetto generale.
  if(roleCap>0) dyn=Math.min(dyn,roleCap);
  dyn=Math.min(dyn,maxAffordable(me));
  return Math.max(0,dyn);
