x}`,help:`FVM ${p.fvm} → riferimento su 600: ${p.base600}. Inflazione ${p.role}: ${fmtPct(inflation(p.role))}.`};
  if(bid<=p.base600) return {cls:"good",txt:`RILANCIA — sotto il target (${p.base600})`,help:`Tetto dinamico attuale: ${max}.`};
  if(bid<=max) return {cls:"warn",txt:`VALUTA — non superare ${max}`,help:`Se il prezzo sale oltre ${max}, il costo compromette il piano attuale.`};
  return {cls:"bad",txt:`LASCIA — oltre il massimo (${max})`,help:`Prezzo corrente ${bid}; target FVM su 600: ${p.base600}.`};
}
